# Worker configuration for the founders-click app, checked before deploy by
# .github/workflows/deploy-app.yml against `wrangler secret list`.
#
# TWO SECTIONS, and the difference matters:
#
#   [required]     No default in code. The feature is broken or blocked
#                  without it. A missing one FAILS the deploy.
#   [recommended]  Code has a working fallback. A missing one WARNS.
#
# The first version of this file listed everything as required, including four
# names with perfectly good defaults, so it would have blocked deploys over
# configuration the app did not actually need. Verified against the code with
# `grep -rhoE "process\.env\.[A-Z_0-9]+" src/` before splitting.
#
# Set with `wrangler secret put <NAME>`. Secrets persist across deploys, so CI
# only verifies they exist and never handles a value.
#
# NOTE: `wrangler secret list` shows SECRETS only, not plaintext Text
# variables. A name set as a Text variable in the dashboard will still be
# reported missing here. Set these as Secrets.

[required]
SUPABASE_URL                  project URL
SUPABASE_PUBLISHABLE_KEY      anon/publishable key, used by the auth middleware
SUPABASE_SERVICE_ROLE_KEY     service role; bypasses RLS, server-only.
                              # client.server.ts also accepts SERVICE_ROLE_KEY
                              # as a fallback, but prefer this canonical name.
EMAILIT_API_KEY               transactional email; no default, sending fails without it
CLOUDFLARE_API_TOKEN          PROVISIONING token (custom hostnames + SSL).
                              # NOT the CI deploy token. Without it,
                              # isEdgeProvisioningConfigured() is false and
                              # customer domain setup is refused outright.
CLOUDFLARE_ZONE_ID            founders.click zone; same consequence as above
CRON_SECRET                   shared secret pg_cron presents to the scheduled
                              # hooks. PROMOTED FROM [recommended] 2026-09-14.
                              # The old note claimed "Unset is SAFE — the hooks
                              # fail closed with 401". Both halves were wrong.
                              # /api/public/hooks/sync-sharetribe returns 500
                              # "server misconfigured", not 401; and unset is the
                              # opposite of safe — it silently disables the
                              # every-30-minute Sharetribe sync that the welcome
                              # email promises every customer. Production ran
                              # that way undetected because this file only
                              # WARNED on it. A deploy must now fail instead.

OPENROUTER_API_KEY            platform OpenRouter key for page generation.
                              # Read through the env fallback in
                              # workspace-secrets.server.ts (process.env[name]),
                              # so a literal grep for process.env.OPENROUTER_API_KEY
                              # finds nothing. Without it every workspace that has
                              # not brought its own key (all beta tenants: the
                              # BYOK page is outside launch scope) cannot generate
                              # a single page.
LOVABLE_API_KEY               platform key for ai.gateway.lovable.dev. Read
                              # through the SAME env fallback in
                              # workspace-secrets.server.ts (process.env[name]),
                              # so a literal grep for process.env.LOVABLE_API_KEY
                              # finds nothing in src/ either — which is how it
                              # was left out of this file while three launch
                              # dashboard surfaces depend on it and fail outright
                              # without it: the Daily Briefing "do it" actions
                              # (coach-actions.functions.ts), the page auditor
                              # (admin-page-auditor.functions.ts) and the SEO
                              # coach (admin-seo-coach.functions.ts). The
                              # help-assistant edge functions
                              # (supabase/functions/help-assistant-*) read their
                              # OWN copy from Supabase function secrets, not the
                              # Worker; this preflight does not cover that copy.

[recommended]
SEND_EMAIL_HOOK_SECRET        Supabase Auth send-email hook signing secret.
                              # REQUIRED IN PRACTICE while the Supabase project
                              # has the hook enabled: without it the endpoint
                              # 401s, GoTrue fails the hook call, and SIGNUP
                              # ITSELF breaks platform-wide (observed live
                              # 2026-09-01). Listed as recommended only because
                              # a project with the hook disabled needs nothing.
OPS_PROBE_SECRET              shared secret for the two ops probes,
                              # /api/public/ops/email-probe and
                              # /api/public/ops/sync-health. Unset closes them:
                              # every call answers 401, and the app is otherwise
                              # unaffected. It must be its OWN value, never a
                              # copy of SEND_EMAIL_HOOK_SECRET — that key signs
                              # Auth's send-email hook, and the probes stopped
                              # accepting it on 2026-09-22 so a curl in a runbook
                              # or a CI log could never become a copy of the
                              # credential that authenticates signups. Any long
                              # random string; rotate freely.
