# Worker secrets required by the founders-click app in production.
#
# Checked BEFORE deploy by .github/workflows/deploy-app.yml against
# `wrangler secret list`. A missing secret fails the run rather than shipping a
# build that 500s the first time it touches the missing dependency.
#
# Set once with `wrangler secret put <NAME>` — they persist across deploys, so
# CI never handles their values and a workflow log can never contain one.
#
# Lines starting with # are comments. Anything after a name on the same line is
# documentation, not part of the name.

# --- Supabase (data plane) --------------------------------------------------
SUPABASE_URL                  project URL
SUPABASE_PUBLISHABLE_KEY      anon/publishable key, used by the auth middleware
SUPABASE_SERVICE_ROLE_KEY     service role; bypasses RLS, server-only

# --- Cloudflare (customer domain provisioning) ------------------------------
# NOT the Worker deployment token. This one creates custom hostnames, SSL and
# routes, and deliberately has no Workers Scripts permission.
CLOUDFLARE_API_TOKEN          provisioning token
CLOUDFLARE_ZONE_ID            founders.click zone

# --- Email ------------------------------------------------------------------
EMAILIT_API_KEY               transactional send
